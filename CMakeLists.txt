cmake_minimum_required(VERSION 3.24)
project(PitcheeIOS LANGUAGES C CXX)
include(FetchContent)
set(NCNN_VULKAN OFF CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_DISABLE_RTTI ON CACHE BOOL "" FORCE)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ncnn GIT_REPOSITORY https://github.com/Tencent/ncnn.git GIT_TAG c4193aadbbb56582aa87b1850dd3d98fb8fd936d)
FetchContent_MakeAvailable(ncnn)
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/PitcheeCore)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CORE}/src/*.cpp)
add_library(pitchee_core STATIC ${CORE_SOURCES})
target_compile_features(pitchee_core PUBLIC cxx_std_17)
target_compile_definitions(pitchee_core PUBLIC PITCHEE_STATIC=1 PRIVATE PITCHEE_CORE_BUILD=1 PITCHEE_ENABLE_NCNN=1)
target_compile_options(pitchee_core PRIVATE -fno-rtti)
target_include_directories(pitchee_core PUBLIC ${CORE}/include PRIVATE ${CORE}/src ${ncnn_SOURCE_DIR}/src/.. ${CMAKE_BINARY_DIR}/ncnn-include)
# Core uses <ncnn/net.h>; ncnn's build tree exposes unprefixed headers.
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/ncnn-include)
file(CREATE_LINK ${ncnn_SOURCE_DIR}/src ${CMAKE_BINARY_DIR}/ncnn-include/ncnn SYMBOLIC)
target_link_libraries(pitchee_core PRIVATE ncnn)
install(TARGETS pitchee_core ARCHIVE DESTINATION lib)
